  <script>
    const quizQuestions = [
      { question: "What is the value of 3² + 4²?", options: ["7", "12", "25", "49"], answer: 2 },
      { question: "If f(x) = 2x² - 3x + 1, what is f(2)?", options: ["3", "5", "7", "9"], answer: 2 },
      { question: "In the equation ax + b = 0, what is the value of x?", options: ["x = a", "x = -b/a", "x = b/a", "x = -a/b"], answer: 1 },
      { question: "Which law states that F = ma?", options: ["Law of inertia", "Law of acceleration", "Law of action-reaction", "Law of gravity"], answer: 1 },
      { question: "What is the SI unit of force?", options: ["Joule", "Watt", "Newton", "Pascal"], answer: 2 },
      { question: "Which quantity measures the rate of change of velocity?", options: ["Displacement", "Speed", "Velocity", "Acceleration"], answer: 3 },
      { question: "What is the chemical symbol of gold?", options: ["Ag", "Au", "Fe", "Cu"], answer: 1 },
      { question: "In a neutral atom, the number of protons equals the number of:", options: ["Neutrons", "Electrons", "Nucleons", "Ions"], answer: 1 },
      { question: "Which of the following is a noble gas?", options: ["Oxygen", "Hydrogen", "Helium", "Nitrogen"], answer: 2 },
      { question: "Which organelle is known as the 'powerhouse of the cell'?", options: ["Nucleus", "Ribosome", "Mitochondria", "Lysosome"], answer: 2 },
      { question: "What process converts glucose and oxygen into energy in cells?", options: ["Photosynthesis", "Fermentation", "Cellular respiration", "Transpiration"], answer: 2 },
      { question: "Which biomolecule stores genetic information?", options: ["Protein", "Carbohydrate", "Lipid", "Nucleic acid"], answer: 3 },
      { question: "Which senior-high track focuses on 'Science, Technology, Engineering, and Mathematics'?", options: ["HUMSS", "ABM", "TVL", "STEM"], answer: 3 },
      { question: "Which of the following is NOT a specialized subject in the STEM strand?", options: ["Pre-Calculus", "Basic Calculus", "General Biology 1", "Cookery"], answer: 3 },
      { question: "Which Filipino scientist is known for his work on crop science and ‘father of green revolution’?", options: ["Maria Orosa", "Ramon Magsaysay", "Emil Javier", "Ramon Barba"], answer: 2 }
    ];

    let current = 0;
    let score = 0;

    const gameArea = document.getElementById("gameArea");
    const finalResult = document.getElementById("finalResult");
    const scoreElem = document.getElementById("score");
    const totalElem = document.getElementById("total");
    const questionText = document.getElementById("questionText");
    const optionsElem = document.getElementById("options");
    const feedbackElem = document.getElementById("feedback");
    const nextBtn = document.getElementById("nextBtn");
    const retryBtn = document.getElementById("retryBtn");

    function startQuiz() {
      current = 0;
      score = 0;
      scoreElem.textContent = "0";
      totalElem.textContent = quizQuestions.length;
      gameArea.style.display = "block";
      finalResult.style.display = "none";
      loadQuestion();
    }

    function loadQuestion() {
      const q = quizQuestions[current];
      questionText.textContent = q.question;
      optionsElem.innerHTML = "";
      feedbackElem.style.display = "none";
      nextBtn.disabled = true;

      q.options.forEach((opt, i) => {
        const btn = document.createElement("button");
        btn.className = "option";
        btn.innerHTML = `<span style="margin-right:8px">🧪</span>${opt}`;
        btn.onclick = () => handleAnswer(i);
        optionsElem.appendChild(btn);
      });
    }

    function handleAnswer(index) {
      const q = quizQuestions[current];
      const correct = index === q.answer;
      
      Array.from(optionsElem.children).forEach(btn => btn.disabled = true);
      
      feedbackElem.style.display = "block";
      if (correct) {
        score++;
        feedbackElem.className = "feedback correct";
        feedbackElem.innerHTML = "✅ Correct!";
      } else {
        feedbackElem.className = "feedback incorrect";
        feedbackElem.innerHTML = `❌ Incorrect. The answer is: ${q.options[q.answer]}`;
      }
      
      scoreElem.textContent = score;
      nextBtn.disabled = false;
    }

    nextBtn.onclick = () => {
      current++;
      if (current < quizQuestions.length) {
        loadQuestion();
      } else {
        showFinal();
      }
    };

    function showFinal() {
      gameArea.style.display = "none";
      finalResult.style.display = "block";
      
      const percent = Math.round((score / quizQuestions.length) * 100);
      document.getElementById("finalScoreVal").textContent = `${score} / ${quizQuestions.length}`;
      document.getElementById("finalPercent").textContent = `Score Percentage: ${percent}%`;

      let msg, icon;
      if (score === 15) {
        msg = "Perfect! You're thinking like a future scientist or engineer!";
        icon = "🏆";
      } else if (score >= 12) {
        msg = "Excellent! You have a very strong grasp of STEM subjects.";
        icon = "🚀";
      } else {
        msg = "Good effort! Review your notes and try for a perfect score!";
        icon = "🔬";
      }

      document.getElementById("messageBox").innerHTML = `<h1>${icon}</h1><strong>${msg}</strong>`;
    }

    // This is the manual reset function
    retryBtn.onclick = () => {
      startQuiz();
    };

    // Initialize the quiz on page load
    startQuiz();
  </script>
